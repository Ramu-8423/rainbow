<script>
// Function to show image preview
function showImage(el) {
  const imgSrc = el.getAttribute('src');
  const preview = document.getElementById('imagePreview');
  const previewImg = document.getElementById('previewImg');
  previewImg.src = imgSrc;
  preview.classList.add('active');
  document.body.style.overflow = 'hidden';
}

// Function to hide image preview
function hideImage(event = null) {
  if (event) event.stopPropagation();
  const preview = document.getElementById('imagePreview');
  const previewImg = document.getElementById('previewImg');
  preview.classList.remove('active');
  previewImg.src = '';
  document.body.style.overflow = 'auto';
}

// Function to show flash message
function showUniqueFlashMessage(message, type = 'success', duration = 3000) {
  const container = document.getElementById('uniqueFlashContainer');
  if (!container) return;

  const msgId = 'flashMsg_' + Date.now();
  const msgDiv = document.createElement('div');
  msgDiv.id = msgId;
  msgDiv.className = `unique-flash-msg alert alert-${type} alert-dismissible fade show`;
  msgDiv.style.minWidth = '200px';
  msgDiv.style.marginBottom = '10px';
  msgDiv.innerHTML = `
    ${message}
    <button type="button" class="btn-close" data-bs-dismiss="alert" aria-label="Close"></button>
  `;

  container.appendChild(msgDiv);

  setTimeout(() => {
    const el = document.getElementById(msgId);
    if (el) container.removeChild(el);
  }, duration);
}

// Helper function to extract document status from cell
function getDocStatusFromCell(cell) {
  if (!cell) return null;
  
  const button = cell.querySelector('button');
  if (!button) return null;
  
  const buttonText = button.textContent.trim();
  
  if (buttonText === 'Rejected') return 0;
  if (buttonText === 'Submitted') return 1;
  if (buttonText === 'Not Filled') return 2;
  
  return null;
}

    <div class="mb-2"><strong>Selfie:</strong> ${cells[18].querySelector('img') ? `<img src="${cells[18].querySelector('img').src}" class="document-img" onclick="showImage(this)">` : 'N/A'}</div>
  `;
  
  // Populate Driver Details
  const driverDetails = document.getElementById('driverDetails');
  driverDetails.innerHTML = `
    <div class="mb-2"><strong>Driver Name:</strong> ${cells[19].textContent}</div>
    <div class="mb-2"><strong>License Back:</strong> ${cells[20].querySelector('img') ? `<img src="${cells[20].querySelector('img').src}" class="document-img" onclick="showImage(this)">` : 'N/A'}</div>
    <div class="mb-2"><strong>License Front:</strong> ${cells[21].querySelector('img') ? `<img src="${cells[21].querySelector('img').src}" class="document-img" onclick="showImage(this)">` : 'N/A'}</div>
    <div class="mb-2"><strong>Phone:</strong> ${cells[22].textContent}</div>
    <div class="mb-2"><strong>Operator Type:</strong> ${cells[23].textContent}</div>
  `;
  
  // Reset all inputs first
  document.querySelectorAll('.reject-check, .verify-check').forEach(radio => radio.checked = false);
  document.querySelectorAll('textarea').forEach(ta => {
    ta.value = '';
    ta.classList.add('d-none');
  });
  
  // Auto-check based on current status
  if (isRejected) {
    if (vehicleDocStatus === 0) {
      document.getElementById('vehicleReject').checked = true;
      document.getElementById('vehicleReason').classList.remove('d-none');
      document.getElementById('vehicleReason').value = rejectionReason;
    }
    
    if (ownerDocStatus === 0) {
      document.getElementById('ownerReject').checked = true;
      document.getElementById('ownerReason').classList.remove('d-none');
      document.getElementById('ownerReason').value = rejectionReason;
    }
    
    if (driverDocStatus === 0) {
      document.getElementById('driverReject').checked = true;
      document.getElementById('driverReason').classList.remove('d-none');
      document.getElementById('driverReason').value = rejectionReason;
    }
  }
  
  if (isVerified) {
    if (vehicleDocStatus === 1) document.getElementById('vehicleVerify').checked = true;
    if (ownerDocStatus === 1) document.getElementById('ownerVerify').checked = true;
    if (driverDocStatus === 1) document.getElementById('driverVerify').checked = true;
  }
  
  // Auto-select specific document type if provided
  if (docType) {
    const docTypeMap = {
      'vehicle': { reject: 'vehicleReject', verify: 'vehicleVerify', reason: 'vehicleReason' },
      'owner': { reject: 'ownerReject', verify: 'ownerVerify', reason: 'ownerReason' },
      'driver': { reject: 'driverReject', verify: 'driverVerify', reason: 'driverReason' }
    };
    
    const elements = docTypeMap[docType];
    if (elements) {
      const docStatus = docType === 'vehicle' ? vehicleDocStatus : 
                       docType === 'owner' ? ownerDocStatus : driverDocStatus;
      
      if (docStatus === 0) {
        document.getElementById(elements.reject).checked = true;
        document.getElementById(elements.reason).classList.remove('d-none');
        document.getElementById(elements.reason).value = rejectionReason;
      } else if (docStatus === 1) {
        document.getElementById(elements.verify).checked = true;
      }
    }
  }
  
  // Show modal using Bootstrap
  const modalElement = document.getElementById('verifyModal');
  const modal = new bootstrap.Modal(modalElement);
  modal.show();
}

// DOM Ready
document.addEventListener('DOMContentLoaded', function() {
  console.log('DOM loaded - initializing event listeners');
  
  // Handle verify_document dropdown change
  document.querySelectorAll('.verify-dropdown').forEach(drop => {
    drop.addEventListener('change', function() {
      const driverId = this.getAttribute('data-driver-id');
      const status = this.value;

      if (status == '1') { // Pending
        updateVerifyStatus(driverId, status);
      } else { // Verified or Rejected - open modal
        openVerifyModal(driverId);
      }
    });
  });

  // Show/hide reason textarea when reject is checked
  document.querySelectorAll('.reject-check').forEach(radio => {
    radio.addEventListener('change', function() {
      const reasonBox = this.closest('.col-md-4').querySelector('textarea');
      if (this.checked && this.value === 'reject') {
        reasonBox.classList.remove('d-none');
      } else {
        reasonBox.classList.add('d-none');
        reasonBox.value = '';
      }
    });
  });

  // Hide reason textarea when verify is checked
  document.querySelectorAll('.verify-check').forEach(radio => {
    radio.addEventListener('change', function() {
      const reasonBox = this.closest('.col-md-4').querySelector('textarea');
      if (this.checked) {
        reasonBox.classList.add('d-none');
        reasonBox.value = '';
      }
    });
  });

  // Submit button click
  document.getElementById('submitVerifyModal').addEventListener('click', function() {
    const driverId = document.getElementById('verifyDriverId').value;

    const selections = {
      vehicle: document.querySelector('input[name="vehicleAction"]:checked')?.value || null,
      owner: document.querySelector('input[name="ownerAction"]:checked')?.value || null,
      driver: document.querySelector('input[name="driverAction"]:checked')?.value || null,
      vehicleReason: document.getElementById('vehicleReason').value.trim(),
      ownerReason: document.getElementById('ownerReason').value.trim(),
      driverReason: document.getElementById('driverReason').value.trim()
    };

    // Validate at least one action is selected
    if (!selections.vehicle && !selections.owner && !selections.driver) {
      showUniqueFlashMessage('Please select at least one action (Verify/Reject) for any section', 'warning');
      return;
    }

    // Validate rejection reasons
    if ((selections.vehicle === 'reject' && !selections.vehicleReason) ||
        (selections.owner === 'reject' && !selections.ownerReason) ||
        (selections.driver === 'reject' && !selections.driverReason)) {
      showUniqueFlashMessage('Please provide rejection reasons for all rejected sections', 'warning');
      return;
    }

    // Determine overall status
    let overallStatus = 1; // Default to pending
    if (selections.vehicle === 'reject' || selections.owner === 'reject' || selections.driver === 'reject') {
      overallStatus = 3; // Any rejected
    } else if (selections.vehicle === 'verify' && selections.owner === 'verify' && selections.driver === 'verify') {
      overallStatus = 2; // All verified
    }

    fetch('/driver/update-multi-verify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ driverId, ...selections, overallStatus })
    })
    .then(res => res.json())
    .then(data => {
      if (data.success) {
        showUniqueFlashMessage(data.message || 'Status updated successfully!', 'success');
        
        // Update dropdown value
        const select = document.querySelector(`.verify-dropdown[data-driver-id="${driverId}"]`);
        if (select) select.value = overallStatus;
        
        // Close modal
        const modal = bootstrap.Modal.getInstance(document.getElementById('verifyModal'));
        if (modal) modal.hide();
        
        // Reload page after 1 second to reflect changes
        setTimeout(() => {
          window.location.reload();
        }, 1000);
      } else {
        showUniqueFlashMessage(data.message || 'Failed to update status', 'danger');
      }
    })
    .catch(err => {
      console.error(err);
      showUniqueFlashMessage('Server error while updating status', 'danger');
    });
  });
});

function updateVerifyStatus(driverId, status) {
  fetch('/driver/update-verify-status', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ driverId, status })
  })
  .then(res => res.json())
  .then(data => {
    if (data.success) {
      showUniqueFlashMessage('Status updated to Pending!', 'success');
      setTimeout(() => {
        window.location.reload();
      }, 1000);
    } else {
      showUniqueFlashMessage(data.error || 'Failed to update status', 'danger');
    }
  })
  .catch(err => {
    console.error(err);
    showUniqueFlashMessage('Server error while updating status', 'danger');
  });
}
